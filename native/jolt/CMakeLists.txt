cmake_minimum_required(VERSION 3.20)
project(AiNativeJolt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED JOLT_SOURCE_DIR)
  message(FATAL_ERROR "Pass pinned JOLT_SOURCE_DIR from scripts/build_native.py")
endif()
foreach(flag INTERPROCEDURAL_OPTIMIZATION DOUBLE_PRECISION USE_ASSERTS GENERATE_DEBUG_SYMBOLS CPP_RTTI_ENABLED
    FLOATING_POINT_EXCEPTIONS_ENABLED ENABLE_ALL_WARNINGS ENABLE_INSTALL ENABLE_OBJECT_STREAM
    DEBUG_RENDERER_IN_DEBUG_AND_RELEASE PROFILER_IN_DEBUG_AND_RELEASE
    JPH_USE_DX12 JPH_USE_VK JPH_USE_MTL JPH_USE_CPU_COMPUTE
    TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER
    DEBUG_RENDERER_IN_DISTRIBUTION PROFILER_IN_DISTRIBUTION JPH_BUILD_SHARED JPH_INSTALL JPH_SAMPLES
    USE_AVX USE_AVX2 USE_AVX512 USE_F16C USE_FMADD USE_LZCNT USE_TZCNT)
  set(${flag} OFF CACHE BOOL "Locked bridge configuration" FORCE)
endforeach()
set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "Stable floating point operations; platform compatibility still requires evidence" FORCE)
set(CPP_EXCEPTIONS_ENABLED ON CACHE BOOL "Catch failures at the C boundary" FORCE)
set(OVERRIDE_CXX_FLAGS OFF CACHE BOOL "Use parent toolchain" FORCE)
set(USE_STATIC_MSVC_RUNTIME_LIBRARY OFF CACHE BOOL "Use platform runtime" FORCE)
set(JOLT_PHYSICS_ROOT "${JOLT_SOURCE_DIR}")
add_subdirectory("${JOLTC_SOURCE_DIR}" joltc-build)
get_target_property(joltc_sources joltc SOURCES)
list(FILTER joltc_sources EXCLUDE REGEX "joltc\\.cpp$")
set_property(TARGET joltc PROPERTY SOURCES "${joltc_sources}")
target_sources(joltc PRIVATE "${JOLTC_EXTENDED_SOURCE}")
target_include_directories(joltc PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${JOLTC_SOURCE_DIR}/src")
if(NOT EXISTS "${JOLT_BODY_MANAGER}")
  message(FATAL_ERROR "Build through scripts/build_native.py to apply restore exception safety")
endif()
get_target_property(jolt_sources Jolt SOURCES)
list(FILTER jolt_sources EXCLUDE REGEX "(BodyManager|JobSystemThreadPool)\\.cpp$")
set_property(TARGET Jolt PROPERTY SOURCES "${jolt_sources}")
target_sources(Jolt PRIVATE "${JOLT_BODY_MANAGER}" "${JOLT_THREAD_POOL}")
add_executable(thread_start_probe thread_start_probe.cpp bridge.cpp "${JOLT_THREAD_POOL}")
target_compile_definitions(thread_start_probe PRIVATE AINATIVE_THREAD_START_TEST=1)
target_link_libraries(thread_start_probe PRIVATE joltc Jolt)
add_library(ainative_jolt SHARED bridge.cpp)
target_link_libraries(ainative_jolt PRIVATE joltc Jolt)
set_target_properties(ainative_jolt PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  # joltc C entry points may throw before our outer C ABI catches the error.
  # /EHsc assumes extern C calls cannot throw and can omit required unwinding.
  foreach(target ainative_jolt joltc thread_start_probe)
    target_compile_options(${target} PRIVATE /fp:precise /EHs /EHc- /GR-)
  endforeach()
else()
  foreach(target ainative_jolt joltc thread_start_probe)
    target_compile_options(${target} PRIVATE -ffp-contract=off -fno-rtti)
  endforeach()
endif()
