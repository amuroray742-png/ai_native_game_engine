cmake_minimum_required(VERSION 3.20)
project(AiNativeJolt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED JOLT_SOURCE_DIR)
  message(FATAL_ERROR "Pass pinned JOLT_SOURCE_DIR from scripts/build_native.py")
endif()
foreach(flag INTERPROCEDURAL_OPTIMIZATION DOUBLE_PRECISION USE_ASSERTS GENERATE_DEBUG_SYMBOLS
    FLOATING_POINT_EXCEPTIONS_ENABLED ENABLE_ALL_WARNINGS ENABLE_INSTALL ENABLE_OBJECT_STREAM
    DEBUG_RENDERER_IN_DEBUG_AND_RELEASE PROFILER_IN_DEBUG_AND_RELEASE
    JPH_USE_DX12 JPH_USE_VK JPH_USE_MTL JPH_USE_CPU_COMPUTE
    TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER
    USE_AVX USE_AVX2 USE_AVX512 USE_F16C USE_FMADD USE_LZCNT USE_TZCNT)
  set(${flag} OFF CACHE BOOL "Locked bridge configuration" FORCE)
endforeach()
set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "Stable floating point operations; platform compatibility still requires evidence" FORCE)
set(CPP_EXCEPTIONS_ENABLED ON CACHE BOOL "Catch failures at the C boundary" FORCE)
set(OVERRIDE_CXX_FLAGS OFF CACHE BOOL "Use parent toolchain" FORCE)
set(USE_STATIC_MSVC_RUNTIME_LIBRARY OFF CACHE BOOL "Use platform runtime" FORCE)
add_subdirectory("${JOLT_SOURCE_DIR}/Build" jolt-build)
if(NOT EXISTS "${JOLT_BODY_MANAGER}")
  message(FATAL_ERROR "Build through scripts/build_native.py to apply restore exception safety")
endif()
get_target_property(jolt_sources Jolt SOURCES)
list(FILTER jolt_sources EXCLUDE REGEX "BodyManager\\.cpp$")
set_property(TARGET Jolt PROPERTY SOURCES "${jolt_sources}")
target_sources(Jolt PRIVATE "${JOLT_BODY_MANAGER}")
add_library(ainative_jolt SHARED bridge.cpp)
target_link_libraries(ainative_jolt PRIVATE Jolt)
set_target_properties(ainative_jolt PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(ainative_jolt PRIVATE /fp:precise /EHsc)
else()
  target_compile_options(ainative_jolt PRIVATE -ffp-contract=off)
endif()
